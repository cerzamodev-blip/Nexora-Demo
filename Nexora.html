📋 Tablas SQL requeridas
-- Ejecuta esto en Supabase SQL Editor:
CREATE TABLE IF NOT EXISTS config (id text PRIMARY KEY, data jsonb, updated_at timestamptz);
CREATE TABLE IF NOT EXISTS clientes (id text PRIMARY KEY, nombre text, rnc text, tel text, email text, direccion text, ruta text, credito_dias int, notas text, activo bool, ts bigint, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS facturas (id text PRIMARY KEY, numero text, cliente_id text, cliente_nm text, cliente_rnc text, cliente_razon text, fecha date, vence date, items jsonb, subtotal numeric, descuento numeric, itbis numeric, itbis_pct int, itbis_activo bool, impuesto numeric, total numeric, ncf text, ncf_tipo text, pagado numeric, saldo numeric, estado text, notas text, ts bigint, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS cobros (id text PRIMARY KEY, factura_id text, factura_num text, cliente_id text, cliente_nm text, monto numeric, metodo text, ref text, fecha date, notas text, chofer_id text, chofer_nm text, ts bigint, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS productos (id text PRIMARY KEY, nombre text, descripcion text, unidad text, precio numeric, activo bool, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS rutas (id text PRIMARY KEY, nombre text, descripcion text, dias text, chofer text, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS gastos (id text PRIMARY KEY, desc text, cat text, monto numeric, fecha date, proveedor text, rnc_proveedor text, metodo text, ncf_activo bool, ncf_tipo text, ncf text, notas text, ts bigint, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS usuarios (id text PRIMARY KEY, nombre text, rol text, pin text, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS gps_tracks (id text PRIMARY KEY, chofer_id text, chofer_nm text, lat numeric, lng numeric, ts bigint, created_at timestamptz DEFAULT now(), updated_at timestamptz);
CREATE TABLE IF NOT EXISTS pedidos (id text PRIMARY KEY, ruta_id text, cliente_id text, cliente_nm text, cliente_dir text, cliente_tel text, fecha date, items jsonb, subtotal numeric, total numeric, modalidad text, metodo text, dias_credito int, vence date, estado text, notas text, ts bigint, created_at timestamptz DEFAULT now(), updated_at timestamptz);
